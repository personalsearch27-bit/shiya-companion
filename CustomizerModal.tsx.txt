import React, { useRef } from 'react';
import { CharacterProfile, MoodType } from '../types';
import { X, Upload, RotateCcw } from 'lucide-react';

interface Props {
  isOpen: boolean;
  onClose: () => void;
  profile: CharacterProfile;
  onSaveProfile: (profile: CharacterProfile) => void;
  onUploadAvatar: (base64: string) => void;
  onResetAvatar: () => void;
  onTestMood: (mood: MoodType) => void;
}

export const CustomizerModal: React.FC<Props> = ({
  isOpen,
  onClose,
  profile,
  onSaveProfile,
  onUploadAvatar,
  onResetAvatar,
  onTestMood,
}) => {
  const fileInputRef = useRef<HTMLInputElement | null>(null);

  if (!isOpen) return null;

  const handleFileChange = (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0];
    if (!file) return;

    if (!file.type.startsWith('image/')) {
      alert('Please select an image file (PNG, JPG, WebP).');
      return;
    }

    const reader = new FileReader();
    reader.onload = () => {
      if (typeof reader.result === 'string') {
        onUploadAvatar(reader.result);
      }
    };
    reader.readAsDataURL(file);
  };

  return (
    <div className="fixed inset-0 z-50 bg-black/75 backdrop-blur-sm flex items-center justify-center p-4">
      <div className="bg-[#141219] border border-white/10 w-full max-w-md rounded-2xl p-5 shadow-2xl flex flex-col max-h-[90vh] overflow-y-auto">
        <div className="flex items-center justify-between pb-3 border-b border-white/10">
          <h2 className="text-sm font-semibold text-white tracking-wide">Avatar & Appearance</h2>
          <button onClick={onClose} className="p-1 rounded-full text-[#A49BAA] hover:text-white">
            <X size={18} />
          </button>
        </div>

        <div className="space-y-4 py-4 text-xs">
          {/* Avatar Upload */}
          <div>
            <label className="block text-[#A49BAA] mb-2 font-medium">Custom Character Art</label>
            <div className="flex gap-2">
              <input
                type="file"
                ref={fileInputRef}
                onChange={handleFileChange}
                accept="image/*"
                className="hidden"
              />
              <button
                type="button"
                onClick={() => fileInputRef.current?.click()}
                className="flex-1 flex items-center justify-center gap-2 py-2.5 px-3 rounded-xl bg-[#1D1922] border border-white/10 hover:border-[#FF304F]/50 text-white transition active:scale-98"
              >
                <Upload size={14} /> Upload Custom Portrait
              </button>
              {profile.customAvatarUrl && (
                <button
                  type="button"
                  onClick={onResetAvatar}
                  className="p-2.5 rounded-xl bg-[#1D1922] border border-white/10 text-rose-400 hover:bg-rose-500/10"
                  title="Reset to Starter Vector Avatar"
                >
                  <RotateCcw size={14} />
                </button>
              )}
            </div>
            <p className="text-[10px] text-[#A49BAA] mt-1.5">
              Saved locally via high-capacity IndexedDB. Works completely offline.
            </p>
          </div>

          {/* Animation Intensity */}
          <div>
            <div className="flex justify-between text-[#A49BAA] mb-1">
              <span>Breathing & Movement Speed</span>
              <span className="text-white">{profile.animationIntensity}x</span>
            </div>
            <input
              type="range"
              min="0.5"
              max="1.5"
              step="0.1"
              value={profile.animationIntensity}
              onChange={(e) => onSaveProfile({ ...profile, animationIntensity: parseFloat(e.target.value) })}
              className="w-full accent-[#FF304F]"
            />
          </div>

          {/* Test Facial Expressions */}
          <div>
            <label className="block text-[#A49BAA] mb-2 font-medium">Preview Expressions</label>
            <div className="grid grid-cols-4 gap-1.5">
              {(['happy', 'shy', 'curious', 'sad', 'sleepy'] as MoodType[]).map((m) => (
                <button
                  key={m}
                  onClick={() => onTestMood(m)}
                  className="py-1.5 px-2 rounded-lg bg-[#1D1922] border border-white/10 hover:border-[#FF304F]/40 capitalize text-center text-[#FAF7FA]"
                >
                  {m}
                </button>
              ))}
            </div>
          </div>
        </div>

        <button
          onClick={onClose}
          className="w-full py-2.5 rounded-xl bg-[#FF304F] text-white font-semibold text-xs active:scale-98 transition shadow-[0_0_15px_rgba(255,48,79,0.3)]"
        >
          Done
        </button>
      </div>
    </div>
  );
};
