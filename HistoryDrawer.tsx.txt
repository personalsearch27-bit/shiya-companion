import React from 'react';
import { ConversationSession } from '../types';
import { X, Plus, Trash2, Download } from 'lucide-react';

interface Props {
  isOpen: boolean;
  onClose: () => void;
  sessions: ConversationSession[];
  activeId: string;
  onSelectSession: (id: string) => void;
  onNewSession: () => void;
  onDeleteSession: (id: string) => void;
  onExportSession: (session: ConversationSession) => void;
}

export const HistoryDrawer: React.FC<Props> = ({
  isOpen,
  onClose,
  sessions,
  activeId,
  onSelectSession,
  onNewSession,
  onDeleteSession,
  onExportSession,
}) => {
  if (!isOpen) return null;

  return (
    <div className="fixed inset-0 z-50 bg-black/75 backdrop-blur-sm flex justify-end">
      <div className="w-full max-w-xs bg-[#141219] h-full border-l border-white/10 p-4 flex flex-col animate-slideLeft">
        <div className="flex items-center justify-between pb-3 border-b border-white/10">
          <h2 className="text-sm font-semibold text-white">Conversations</h2>
          <button onClick={onClose} className="p-1 rounded-full text-[#A49BAA] hover:text-white">
            <X size={18} />
          </button>
        </div>

        <button
          onClick={() => {
            onNewSession();
            onClose();
          }}
          className="mt-3 flex items-center justify-center gap-1.5 py-2.5 px-3 rounded-xl bg-[#FF304F] text-white text-xs font-semibold shadow-[0_0_12px_rgba(255,48,79,0.3)] active:scale-98 transition"
        >
          <Plus size={15} /> New Conversation
        </button>

        <div className="flex-1 overflow-y-auto space-y-2 mt-4">
          {sessions.map((sess) => {
            const isActive = sess.id === activeId;
            return (
              <div
                key={sess.id}
                onClick={() => {
                  onSelectSession(sess.id);
                  onClose();
                }}
                className={`p-3 rounded-xl border text-xs cursor-pointer transition flex items-center justify-between ${
                  isActive
                    ? 'bg-[#1D1922] border-[#FF304F]/40 text-white'
                    : 'bg-[#09090D]/60 border-white/5 text-[#A49BAA] hover:border-white/20'
                }`}
              >
                <div className="truncate pr-2">
                  <div className="font-medium text-white truncate">{sess.title}</div>
                  <div className="text-[10px] text-[#A49BAA]">
                    {new Date(sess.updatedAt).toLocaleDateString([], { month: 'short', day: 'numeric' })}
                  </div>
                </div>

                <div className="flex items-center gap-1" onClick={(e) => e.stopPropagation()}>
                  <button
                    onClick={() => onExportSession(sess)}
                    title="Export as Text"
                    className="p-1 hover:text-white text-[#A49BAA]"
                  >
                    <Download size={13} />
                  </button>
                  <button
                    onClick={() => onDeleteSession(sess.id)}
                    title="Delete Conversation"
                    className="p-1 hover:text-rose-400 text-[#A49BAA]"
                  >
                    <Trash2 size={13} />
                  </button>
                </div>
              </div>
            );
          })}
        </div>
      </div>
    </div>
  );
};
