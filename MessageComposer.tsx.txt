import React, { useState } from 'react';
import { Send, Mic, MicOff, Volume2, VolumeX } from 'lucide-react';
import { VoiceState } from '../types';

interface Props {
  onSendMessage: (text: string) => void;
  voiceState: VoiceState;
  ttsEnabled: boolean;
  onToggleVoiceMode: () => void;
  onToggleTTS: () => void;
  disabled: boolean;
}

export const MessageComposer: React.FC<Props> = ({
  onSendMessage,
  voiceState,
  ttsEnabled,
  onToggleVoiceMode,
  onToggleTTS,
  disabled,
}) => {
  const [text, setText] = useState('');

  const handleSubmit = (e: React.FormEvent) => {
    e.preventDefault();
    if (!text.trim() || disabled) return;
    onSendMessage(text.trim());
    setText('');
  };

  const isVoiceActive = voiceState === 'listening' || voiceState === 'speaking';

  return (
    <footer className="p-3 bg-[#141219]/90 border-t border-white/10 backdrop-blur-md">
      <form onSubmit={handleSubmit} className="flex items-center gap-2 max-w-4xl mx-auto">
        <button
          type="button"
          onClick={onToggleTTS}
          title={ttsEnabled ? 'Spoken replies enabled' : 'Spoken replies muted'}
          className={`p-2.5 rounded-full transition-colors active:scale-95 ${
            ttsEnabled ? 'bg-[#1D1922] text-[#FF8196] border border-[#FF304F]/30' : 'bg-transparent text-[#A49BAA]'
          }`}
        >
          {ttsEnabled ? <Volume2 size={18} /> : <VolumeX size={18} />}
        </button>

        <div className="relative flex-1">
          <input
            type="text"
            value={text}
            onChange={(e) => setText(e.target.value)}
            placeholder={isVoiceActive ? "Listening to your voice..." : "Message Shiya..."}
            disabled={disabled}
            className="w-full bg-[#09090D] border border-white/10 focus:border-[#FF304F] text-[#FAF7FA] placeholder-[#A49BAA] text-xs sm:text-sm rounded-full py-2.5 pl-4 pr-10 outline-none transition"
          />
        </div>

        <button
          type="button"
          onClick={onToggleVoiceMode}
          title="Voice Conversation"
          className={`p-2.5 rounded-full transition-all active:scale-95 ${
            isVoiceActive
              ? 'bg-[#FF304F] text-white animate-pulse shadow-[0_0_15px_rgba(255,48,79,0.5)]'
              : 'bg-[#1D1922] text-white hover:border-[#FF304F]/40 border border-white/10'
          }`}
        >
          {isVoiceActive ? <MicOff size={18} /> : <Mic size={18} />}
        </button>

        <button
          type="submit"
          disabled={!text.trim() || disabled}
          title="Send Message"
          className="p-2.5 rounded-full bg-[#FF304F] disabled:opacity-40 text-white font-medium hover:bg-[#FF304F]/90 active:scale-95 transition shadow-[0_0_12px_rgba(255,48,79,0.3)]"
        >
          <Send size={18} />
        </button>
      </form>
    </footer>
  );
};
